cmake_minimum_required(VERSION 3.20)
project(lightscaper_raw_engine_current_v1 LANGUAGES CXX)
if(NOT IS_DIRECTORY "${LIGHTSCAPER_LIBRAW_SOURCE_ROOT}")
	message(FATAL_ERROR "The exact authenticated LibRaw source extraction is required")
endif()
file(READ "${LIGHTSCAPER_LIBRAW_SOURCE_ROOT}/Makefile.am" SCAPE_LIBRAW_MAKEFILE)
string(REGEX REPLACE "\\\\\r?\n" " " SCAPE_LIBRAW_MAKEFILE "${SCAPE_LIBRAW_MAKEFILE}")
string(REGEX MATCH "lib_libraw_a_SOURCES[ \t]*=[^\r\n]*" SCAPE_LIBRAW_ASSIGNMENT "${SCAPE_LIBRAW_MAKEFILE}")
string(REGEX MATCHALL "src/([a-z0-9_]+/)*[a-z0-9_]+\\.cpp" SCAPE_LIBRAW_UNITS "${SCAPE_LIBRAW_ASSIGNMENT}")
list(LENGTH SCAPE_LIBRAW_UNITS SCAPE_LIBRAW_UNIT_COUNT)
if(SCAPE_LIBRAW_UNIT_COUNT LESS 50 OR SCAPE_LIBRAW_UNIT_COUNT GREATER 128)
	message(FATAL_ERROR "The fixed LibRaw compilation inventory is invalid")
endif()
list(TRANSFORM SCAPE_LIBRAW_UNITS PREPEND "${LIGHTSCAPER_LIBRAW_SOURCE_ROOT}/")
add_library(lightscaper_raw_engine_current_v1 STATIC src/raw_engine.cpp src/raw_recovery.cpp src/raw_preview.cpp ${SCAPE_LIBRAW_UNITS})
target_include_directories(lightscaper_raw_engine_current_v1 PUBLIC include PRIVATE "${LIGHTSCAPER_LIBRAW_SOURCE_ROOT}")
target_compile_features(lightscaper_raw_engine_current_v1 PUBLIC cxx_std_20)
target_compile_definitions(lightscaper_raw_engine_current_v1 PRIVATE LIBRAW_NODLL=1 LIBRAW_NOTHREADS=1 LIBRAW_NO_IOSTREAMS_DATASTREAM= LIBRAW_MAX_THUMBNAIL_MB=16)
if(MSVC)
	target_compile_options(lightscaper_raw_engine_current_v1 PRIVATE /fp:strict /EHsc)
	foreach(SCAPE_FEATURE USE_JPEG USE_ZLIB USE_LCMS USE_LCMS2 USE_RAWSPEED USE_RAWSPEED3 USE_DNGSDK USE_X3FTOOLS _OPENMP)
		target_compile_options(lightscaper_raw_engine_current_v1 PRIVATE "/U${SCAPE_FEATURE}")
	endforeach()
else()
	target_compile_options(lightscaper_raw_engine_current_v1 PRIVATE -O2 -ffp-contract=off -fno-fast-math -fno-finite-math-only)
	foreach(SCAPE_FEATURE USE_JPEG USE_ZLIB USE_LCMS USE_LCMS2 USE_RAWSPEED USE_RAWSPEED3 USE_DNGSDK USE_X3FTOOLS _OPENMP)
		target_compile_options(lightscaper_raw_engine_current_v1 PRIVATE "-U${SCAPE_FEATURE}")
	endforeach()
endif()
option(LIGHTSCAPER_BUILD_RAW_ENGINE_FIXTURE "Build the private native engine fixture" ON)
if(LIGHTSCAPER_BUILD_RAW_ENGINE_FIXTURE)
	add_executable(lightscaper_raw_engine_fixture tests/engine_fixture.cpp)
	target_link_libraries(lightscaper_raw_engine_fixture PRIVATE lightscaper_raw_engine_current_v1)
endif()

add_executable(lightscaper_photo_raw_current_v1 src/raw_process.cpp ../common/sha256.cpp)
target_include_directories(lightscaper_photo_raw_current_v1 PRIVATE ../common)
target_link_libraries(lightscaper_photo_raw_current_v1 PRIVATE lightscaper_raw_engine_current_v1)
