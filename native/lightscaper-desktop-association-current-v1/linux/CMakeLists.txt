cmake_minimum_required(VERSION 3.20)
project(lightscaper_desktop_association_linux_current_v1 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "The association helper requires target-native Linux.")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|aarch64|arm64)$")
  message(FATAL_ERROR "Only maintained native Linux x64/arm64 are admitted.")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0>=2.58)
add_executable(lightscaper-desktop-association-current-v1 main.cpp ../../framescaper-media-host/src/strict_json.cpp)
target_include_directories(lightscaper-desktop-association-current-v1 PRIVATE ../../framescaper-media-host/src)
set_target_properties(lightscaper-desktop-association-current-v1 PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_compile_options(lightscaper-desktop-association-current-v1 PRIVATE -Wall -Wextra -Werror)
target_link_libraries(lightscaper-desktop-association-current-v1 PRIVATE PkgConfig::GIO)
