cmake_minimum_required(VERSION 3.28)
project(lightscaper_desktop_association_current_v1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
	message(FATAL_ERROR "The Lightscaper default association helper requires native Windows MSVC")
endif()
if(LIGHTSCAPER_ASSOCIATION_NATIVE_TARGET STREQUAL "win-x64")
	if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|x86_64|x64)$" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
		message(FATAL_ERROR "The association helper must target native Windows x64")
	endif()
elseif(LIGHTSCAPER_ASSOCIATION_NATIVE_TARGET STREQUAL "win-arm64")
	if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|aarch64|arm64)$" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
		message(FATAL_ERROR "The association helper must target native Windows ARM64")
	endif()
else()
	message(FATAL_ERROR "Explicit win-x64 or win-arm64 LIGHTSCAPER_ASSOCIATION_NATIVE_TARGET required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(lightscaper-desktop-association-current-v1 main.cpp)
target_compile_features(lightscaper-desktop-association-current-v1 PRIVATE cxx_std_17)
set_target_properties(lightscaper-desktop-association-current-v1 PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(lightscaper-desktop-association-current-v1 PRIVATE
	UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00
	LIGHTSCAPER_ASSOCIATION_TARGET="${LIGHTSCAPER_ASSOCIATION_NATIVE_TARGET}")
target_compile_options(lightscaper-desktop-association-current-v1 PRIVATE /W4 /WX /permissive- /utf-8)
target_link_libraries(lightscaper-desktop-association-current-v1 PRIVATE kernel32 shell32 shlwapi ole32 advapi32 bcrypt)
