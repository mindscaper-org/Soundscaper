cmake_minimum_required(VERSION 3.28)
project(lightscaper_photo_native_current_v1 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(NOT LIGHTSCAPER_NATIVE_TARGET MATCHES "^(linux-x64|linux-arm64|mac-arm64|win-x64|win-arm64)$")
 message(FATAL_ERROR "One exact maintained Photo native target is required")
endif()
if(LIGHTSCAPER_NATIVE_TARGET MATCHES "^linux-")
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Photo Linux target requires the actual Linux toolchain")
 endif()
elseif(LIGHTSCAPER_NATIVE_TARGET STREQUAL "mac-arm64")
 if(NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
  message(FATAL_ERROR "Photo macOS target requires the actual arm64 toolchain")
 endif()
elseif(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "Photo Windows targets require the maintained MSVC/CMake toolchain")
endif()
if(LIGHTSCAPER_NATIVE_TARGET MATCHES "-x64$" AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|x86_64|x64)$")
 message(FATAL_ERROR "Photo target CPU must be x64")
elseif(LIGHTSCAPER_NATIVE_TARGET MATCHES "-arm64$" AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|aarch64|arm64)$")
 message(FATAL_ERROR "Photo target CPU must be arm64")
endif()
# Enable languages before child directories inherit their compile rules.
if(APPLE)
 enable_language(OBJCXX)
 set(CMAKE_OBJCXX_STANDARD 20)
 set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(SCAPE_CONFIGURATION Debug Release RelWithDebInfo MinSizeRel)
 string(TOUPPER "${SCAPE_CONFIGURATION}" SCAPE_CONFIGURATION_UPPER)
 set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${SCAPE_CONFIGURATION_UPPER} "${CMAKE_BINARY_DIR}/bin")
endforeach()
set(LIGHTSCAPER_BUILD_RAW_ENGINE_FIXTURE OFF CACHE BOOL "No test fixture in the shipping Photo closure" FORCE)
add_subdirectory(../lightscaper-raw-engine-current-v1 raw-engine)
set(SCAPE_FS ../common/granted_file_reader_posix.cpp)
if(WIN32)
 set(SCAPE_FS ../common/granted_file_reader_windows.cpp)
endif()
add_executable(lightscaper_photo_granted_filesystem_current_v1
 ../lightscaper-granted-filesystem-current-v1/src/main.cpp
 ../lightscaper-granted-filesystem-current-v1/src/protocol.cpp ${SCAPE_FS})
target_include_directories(lightscaper_photo_granted_filesystem_current_v1 PRIVATE ../common ../lightscaper-granted-filesystem-current-v1/include)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 add_executable(milestone5-native-isolation-launcher src/photo_linux_launcher.c)
 target_sources(lightscaper_photo_raw_current_v1 PRIVATE src/photo_linux_bootstrap.c)
 target_compile_definitions(lightscaper_photo_raw_current_v1 PRIVATE LIGHTSCAPER_PHOTO_LINUX_BOOTSTRAP=1)
 set(SCAPE_PROFILE profiles/linux-v1.json)
 set(SCAPE_BROKER profiles/linux-broker-v1.json)
elseif(APPLE)
 add_executable(milestone5-native-isolation-launcher src/photo_macos_launcher.mm)
 target_link_libraries(milestone5-native-isolation-launcher PRIVATE "-lproc")
 target_sources(lightscaper_photo_raw_current_v1 PRIVATE src/photo_macos_bootstrap.mm)
 target_include_directories(lightscaper_photo_raw_current_v1 PRIVATE src)
 target_compile_definitions(lightscaper_photo_raw_current_v1 PRIVATE LIGHTSCAPER_PHOTO_MACOS_BOOTSTRAP=1)
 target_link_libraries(lightscaper_photo_raw_current_v1 PRIVATE "-lsandbox")
 set(SCAPE_PROFILE ../milestone-5-native-isolation-launcher/profiles/macos-v1.sb)
 set(SCAPE_BROKER profiles/macos-broker-v1.json)
else()
 add_executable(milestone5-native-isolation-launcher src/photo_windows_launcher.cpp)
 target_compile_definitions(milestone5-native-isolation-launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
 target_link_libraries(milestone5-native-isolation-launcher PRIVATE advapi32 onecoreuap userenv)
 set(SCAPE_PROFILE profiles/windows-v1.json)
 set(SCAPE_BROKER profiles/windows-broker-v1.json)
endif()
foreach(SCAPE_TARGET lightscaper_photo_granted_filesystem_current_v1 lightscaper_photo_raw_current_v1 milestone5-native-isolation-launcher)
 if(MSVC)
  target_compile_options(${SCAPE_TARGET} PRIVATE /W4 /WX /permissive- /EHsc /fp:strict)
 else()
  target_compile_options(${SCAPE_TARGET} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
   target_link_options(${SCAPE_TARGET} PRIVATE -Wl,--build-id=none)
  endif()
 endif()
endforeach()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/profiles")
get_filename_component(SCAPE_PROFILE_NAME "${SCAPE_PROFILE}" NAME)
get_filename_component(SCAPE_BROKER_NAME "${SCAPE_BROKER}" NAME)
configure_file("${SCAPE_PROFILE}" "${CMAKE_BINARY_DIR}/profiles/${SCAPE_PROFILE_NAME}" COPYONLY)
configure_file("${SCAPE_BROKER}" "${CMAKE_BINARY_DIR}/profiles/${SCAPE_BROKER_NAME}" COPYONLY)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/photo-native-toolchain.json" CONTENT
 "{\"cmakeVersion\":\"${CMAKE_VERSION}\",\"generator\":\"${CMAKE_GENERATOR}\",\"compiler\":\"${CMAKE_CXX_COMPILER_ID}\",\"compilerVersion\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"platform\":\"${CMAKE_SYSTEM_NAME}\",\"architecture\":\"${CMAKE_SYSTEM_PROCESSOR}\"}\n")

option(LIGHTSCAPER_PHOTO_BUILD_QUALIFICATION "Build the private OS-isolation canary; never stage it" OFF)
if(LIGHTSCAPER_PHOTO_BUILD_QUALIFICATION)
 add_executable(lightscaper_photo_isolation_canary_current_v1 tests/isolation_canary.cpp)
 if(APPLE)
  target_sources(lightscaper_photo_isolation_canary_current_v1 PRIVATE src/photo_macos_bootstrap.mm)
  target_include_directories(lightscaper_photo_isolation_canary_current_v1 PRIVATE src)
  target_compile_definitions(lightscaper_photo_isolation_canary_current_v1 PRIVATE LIGHTSCAPER_PHOTO_MACOS_BOOTSTRAP=1)
  target_link_libraries(lightscaper_photo_isolation_canary_current_v1 PRIVATE "-lsandbox")
 elseif(WIN32)
  target_compile_definitions(lightscaper_photo_isolation_canary_current_v1 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(lightscaper_photo_isolation_canary_current_v1 PRIVATE ws2_32)
 else()
  target_sources(lightscaper_photo_isolation_canary_current_v1 PRIVATE src/photo_linux_bootstrap.c)
  target_compile_definitions(lightscaper_photo_isolation_canary_current_v1 PRIVATE LIGHTSCAPER_PHOTO_LINUX_BOOTSTRAP=1)
  target_link_options(lightscaper_photo_isolation_canary_current_v1 PRIVATE -Wl,--build-id=none)
 endif()
endif()
