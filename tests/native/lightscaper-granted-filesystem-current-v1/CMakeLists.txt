cmake_minimum_required(VERSION 3.20)
project(lightscaper_granted_filesystem_fixture LANGUAGES CXX)
get_filename_component(SCAPE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(SCAPE_ROLE_ROOT "${SCAPE_ROOT}/native/lightscaper-granted-filesystem-current-v1")
set(SCAPE_PLATFORM "${SCAPE_ROOT}/native/common/granted_file_reader_posix.cpp")
if(WIN32)
	set(SCAPE_PLATFORM "${SCAPE_ROOT}/native/common/granted_file_reader_windows.cpp")
endif()
set(SCAPE_ENTRY bootstrap.cpp)
if(EXISTS "${SCAPE_ROLE_ROOT}/src/main.cpp")
	set(SCAPE_ENTRY "${SCAPE_ROLE_ROOT}/src/main.cpp")
endif()
set(SCAPE_SOURCES "${SCAPE_PLATFORM}" "${SCAPE_ENTRY}")
foreach(SCAPE_SOURCE protocol.cpp)
	if(EXISTS "${SCAPE_ROLE_ROOT}/src/${SCAPE_SOURCE}")
		list(APPEND SCAPE_SOURCES "${SCAPE_ROLE_ROOT}/src/${SCAPE_SOURCE}")
	endif()
endforeach()
function(scape_fs_target target)
	add_executable(${target} ${SCAPE_SOURCES} ${ARGN})
	target_include_directories(${target} PRIVATE "${SCAPE_ROOT}/native/common" "${SCAPE_ROLE_ROOT}/include")
	target_compile_features(${target} PRIVATE cxx_std_20)
	if(MSVC)
		target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
		target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
	else()
		target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
	endif()
endfunction()
scape_fs_target(lightscaper_granted_filesystem_fixture)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
	scape_fs_target(lightscaper_granted_filesystem_close_fault_fixture close_fault.cpp)
	target_link_options(lightscaper_granted_filesystem_close_fault_fixture PRIVATE -Wl,--wrap=close)
endif()
