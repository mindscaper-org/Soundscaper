cmake_minimum_required(VERSION 3.20)
project(lightscaper_granted_file_reader_fixture LANGUAGES CXX)
get_filename_component(SCAPE_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(SCAPE_GRANTED_SOURCE "${SCAPE_REPOSITORY_ROOT}/native/common/granted_file_reader_posix.cpp")
if(WIN32)
	set(SCAPE_GRANTED_SOURCE "${SCAPE_REPOSITORY_ROOT}/native/common/granted_file_reader_windows.cpp")
endif()
add_executable(lightscaper_granted_file_reader_fixture fixture.cpp)
# TDD bootstrap reaches the absent production header; no fallback implementation.
if(EXISTS "${SCAPE_GRANTED_SOURCE}")
	target_sources(lightscaper_granted_file_reader_fixture PRIVATE "${SCAPE_GRANTED_SOURCE}")
endif()
target_include_directories(lightscaper_granted_file_reader_fixture PRIVATE "${SCAPE_REPOSITORY_ROOT}/native/common")
target_compile_features(lightscaper_granted_file_reader_fixture PRIVATE cxx_std_20)
if(MSVC)
	target_compile_options(lightscaper_granted_file_reader_fixture PRIVATE /W4 /WX /EHsc)
	target_compile_definitions(lightscaper_granted_file_reader_fixture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
else()
	target_compile_options(lightscaper_granted_file_reader_fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
